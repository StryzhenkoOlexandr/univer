Algorithm HeapSort(A)
    // Вхід: невідсортований масив A
    // Вихід: відсортований масив A
    
    n ← length(A)
    
    // Фаза 1: Побудова максимальної купи
    for i ← floor(n/2) - 1 down to 0 do
        Sink(A, i, n)
    
    // Фаза 2: Сортування
    for i ← n - 1 down to 1 do
        Swap(A[0], A[i])
        Sink(A, 0, i)

Algorithm Sink(A, i, n)
    // Вхід: масив A, індекс i, розмір купи n
    // Вихід: елемент A[i] занурений на потрібне місце
    
    k ← i
    while true do
        j ← 2*k + 1                    // лівий нащадок
        
        if j ≥ n then
            break                      // вихід за межі масиву
            
        // Знаходимо найбільшого нащадка
        if j + 1 < n and A[j + 1] > A[j] then
            j ← j + 1                  // правий нащадок більший
            
        // Перевіряємо чи потрібно міняти місцями
        if A[k] ≥ A[j] then
            break                      // елемент на своєму місці
        else
            Swap(A[k], A[j])           // занурюємо елемент нижче
            k ← j                      // переходимо до нащадка
